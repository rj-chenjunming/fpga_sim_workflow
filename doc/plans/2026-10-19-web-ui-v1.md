```python
@app.get("/api/baselines")
def baselines() -> dict[str, Any]:
    from services import config_api
    return config_api.list_baselines(P.WORKSPACE_ROOT)

@app.get("/api/env/check")
def env_check(path: str = Query(...)) -> dict[str, Any]:
    from services import config_api
    registry = config_api.load_registry(P.TOOL_ROOT / "environments.json")
    try:
        return config_api.check_env(path, registry=registry, current_env_root=P.ENV_ROOT)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail={"error": str(exc)}) from exc

@app.get("/api/fs/dirs")
def fs_dirs(path: str = Query(...)) -> dict[str, Any]:
    from services import config_api
    registry = config_api.load_registry(P.TOOL_ROOT / "environments.json")
    roots = config_api.allowed_roots(
        registry, P.WORKSPACE_ROOT, settings_path=P.TOOL_DIR / "SETTINGS.yaml")
    try:
        return config_api.list_dirs(path, allowed_roots=roots, registry=registry)
    except PermissionError as exc:
        raise HTTPException(status_code=403, detail={"error": str(exc)}) from exc
    except ValueError as exc:
        raise HTTPException(status_code=400, detail={"error": str(exc)}) from exc
```

> 注意：`P` 目前没有 `TOOL_DIR`；用 `P.TOOL_ROOT / "loop" / "SETTINGS.yaml"`。

派生规则（与 Spec §4.3 一致）：
- 单节点摘要：遍历事件，记录最后一次 `kind/status/ts`、首个 `node_started.ts`、最后 `node_finished.ts`、`node_failed`、`detail.mode`。
- `status` 优先级：`failed` > `waiting`（pending 闸门匹配或 N9 有 running job）> `working`（最后事件是 `node_started`）> `done`（最后是 `node_finished` 且 status 非 failed）> `idle`。
- checker 两个节点取**最近事件**的节点。
- `elapsed_s`：有 started 且有 finish → `finish-start`；只有 started 且 working → `int(now-start)`；否则 `None`。
- `summary`：`run_dir/ARTIFACTS[node]` 存在则取第一行非空（去掉 `#`），截断 120 字符；`N11` 固定 `note="未实现（审计 A10）"`；其余缺失 `"产物未生成"`。
- `mode`：`detail.get("mode")` 或 `detail.get("model")` 有值 → `live`/`standin`（`detail["mode"]` 优先）；否则 `"—"`。
- `evidence`：产物存在时 `[artifact]`。
- 全部结果附 `"authority": "projection"`（在 run_view 层加一次即可）。
